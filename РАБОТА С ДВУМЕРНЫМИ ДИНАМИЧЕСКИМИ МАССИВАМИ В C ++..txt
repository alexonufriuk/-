double** OA_determ_dvmas(double** ar, int m, int n) {
  ar = new double* [m];
  for (int i = 0; i < m; i++)
    ar[i] = new double[n];
  return ar;
}

double** OA_delete_dvmas(double** ar, int m, int n) {
  for (int i = m - 1; i >= 0; i--) {
    delete[] ar[i];
    ar[i] = NULL;
  }
  delete[] ar;
  ar = NULL;
  return ar;
}

void OA_vvod_dvmas(double** ar, int m, int n) {
  int a = 0, b = 1;
  srand((unsigned int)time(0));
  for (int i = 0; i < m; i++)
    for (int j = 0; j < n; j++)
      ar[i][j] = a + 1. * rand() / RAND_MAX * (b - a);
}

void OA_vivod_dvmas(double** ar, int m, int n) {
  cout.setf(ios::fixed);
  for (int i = 0; i < m; i++) {
    cout << "{";
    for (int j = 0; j < n; j++) {
      cout.precision(3); cout.width(5);  cout << ar[i][j];
      if (j != n - 1) cout << ", ";
    }
    cout << "}" << endl;
  }
  cout.unsetf(ios::fixed);
}

double** OA_transform_dvmas(double** ar, int m, int n) {
  double** temp_ar = NULL;
  int col = -1;
  cout << "the number of column you want to add after (in range 0 - " << n - 1 << "): "; cin >> col;
  if(col<0 || col >=n){
      cout<<"Error: Invalid value entered"<<endl;
      srand((unsigned int)time(0));
      col = rand()%n;
      cout<<"Column number = "<<col<<endl;
  } else{
      cout<<"Ñolumn number is selected correctly. "<<endl;
  }
  temp_ar = new double* [m];
  for (int i = 0; i < m; i++)
    temp_ar[i] = new double[n + 1];
  for (int i = 0; i < m; i++) {
    for (int j = 0; j < n + 1; j++) {
      if (j <= col) temp_ar[i][j] = ar[i][j];
      else if (j == col + 1) temp_ar[i][col + 1] = 0;
      else temp_ar[i][j] = ar[i][j - 1];
    }
  }
  for (int i = m - 1; i >= 0; i--) {
    delete[] ar[i];
    ar[i] = NULL;
  }
  delete[] ar;
  ar = NULL;
  return temp_ar;
}

double** OA_square_dvmas(double** ar1, int n, double** ar2, int k) {
  for (int i = 0; i < k; i++)
    for (int j = 0; j < k; j++) {
      if (i > j) ar2[i][j] = ar1[i][j];
      else if (i < j) ar2[i][j] = 0;
      else {
        double mn = ar1[i][j];
        for (int g = j + 1; g < n; g++) {
          if (mn > ar1[i][g]) mn = ar1[i][g];
        }
        ar2[i][j] = mn;
      }
    }
  return ar2;
}

double** OA_modified_dvmas(double** ar, int k) {
    double** temp_ar = OA_determ_dvmas(NULL, k, k);
    for (int i = 0; i < k; i++) {
        for (int j = 0; j < k; j++) {
            temp_ar[i][j] = ar[k - 1 - j][i];
        }
    }
    OA_delete_dvmas(ar, k, k);
    return temp_ar;
}
