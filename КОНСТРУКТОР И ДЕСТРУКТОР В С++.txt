using namespace std;

class vector
{
    int *v;
    int n;
public:
    vector(int);
    vector(int, int*); 
    ~vector();
    void print();
    void sort(bool a);
    void task();
    int min();          
    int max();          
    double ave();   
};

vector::vector(int nn, int *vv)
{
    n = nn;
    v = new int[n];
    for (int i = 0; i < n; i++)
        v[i] = vv[i];
}

int vector::min()
{
    int min = v[0];
    for (int i = 1; i < n; i++)
        if (v[i] < min) min = v[i];
    return min;
}

int vector::max()
{
    int max = v[0];
    for (int i = 1; i < n; i++)
        if (v[i] > max) max = v[i];
    return max;
}

double vector::ave()
{
    int sum = 0;
    for (int i = 0; i < n; i++)
        sum += v[i];
    return (double)sum / n;
}

vector::vector(int nn)
{
    n = nn;
    v = new int[n];
    for (int i = 0; i < n; i++)
       v[i] = -1 + rand() % (10 - (-1) + 1);
}

vector::~vector()
{
    delete[] v;
}

void vector::print()
{
    cout << "{";
    for (int i = 0; i < n; i++)
        cout << setw(4) << v[i];
    cout << " }" << endl;
}

void vector::sort(bool a)
{ 
    for (int i = 0; i < n - 1; i++)
    {
        for (int j = 0; j < n - i - 1; j++)
        {
            if ((a == false && v[j] < v[j + 1]) || (a == true && v[j] == 0))
            {
                int temp = v[j];
                v[j] = v[j + 1];
                v[j + 1] = temp;
            }
        }
    }
}
